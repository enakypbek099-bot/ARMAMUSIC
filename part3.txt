function lrcParse(s){
  var L=[];
  String(s||'').split(/\r?\n/).forEach(function(ln){
    var m,ts=[],last=0;LRT.lastIndex=0;
    while((m=LRT.exec(ln))){ts.push(+m[1]*60+parseFloat(m[2].replace(':','.')));last=LRT.lastIndex}
    if(ts.length){var tx=ln.slice(last).replace(LRW,'').trim();ts.forEach(function(t){L.push({t:t,x:tx})})}
  });
  if(L.length<2)return null;
  L.sort(function(a,b){return a.t-b.t});
  return L;
}

var RVP={'Комната':{size:.6,damp:55,pre:4,width:70,tone:70,mix:30},'Студия':{size:1.1,damp:30,pre:8,width:80,tone:80,mix:30},'Зал':{size:2,damp:45,pre:20,width:85,tone:75,mix:40},'Собор':{size:4.5,damp:65,pre:35,width:100,tone:55,mix:45}};
function rvIR(){
  var sr=ac.sampleRate,n=Math.floor(sr*rv.size),b=ac.createBuffer(2,n,sr),dm=rv.damp/100*.95,w=rv.width/100,dL=b.getChannelData(0),dR=b.getChannelData(1),yL=0,yR=0,i,t,e,a,fi=Math.floor(sr*.003);
  for(i=0;i<n;i++){
    t=i/n;e=Math.pow(10,-3*t)*Math.min(1,i/fi);a=Math.max(.06,1-dm*Math.min(1,t*1.4));
    yL+=((Math.random()*2-1)*e-yL)*a;yR+=((Math.random()*2-1)*e-yR)*a;
    dL[i]=yL;dR[i]=(1-w)*yL+w*yR;
  }
  return b;
}
function rvBuild(src,dst){
  rvC=ac.createConvolver();rvS=ac.createGain();rvPD=ac.createDelay(.25);rvW=ac.createGain();rvW.gain.value=0;
  rvHp=bzBiq('highpass',150,.7);rvLp=bzBiq('lowpass',8000,.7);
  src.connect(rvS);rvS.connect(rvPD);rvPD.connect(rvC);rvC.connect(rvHp);rvHp.connect(rvLp);rvLp.connect(rvW);rvW.connect(dst);
}
function rvApply(now){
  if(!ac||!rvC)return;var t=ac.currentTime;
  rvPD.delayTime.setTargetAtTime(rv.pre/1000,t,.03);
  rvLp.frequency.setTargetAtTime(Math.min(20000,800*Math.pow(25,rv.tone/100)),t,.03);
  rvW.gain.setTargetAtTime(rvOn?rv.mix/100*.8:0,t,.05);
  if(rvOn){
    var key=function(){return rv.size+'|'+rv.damp+'|'+rv.width};
    if(key()!==rvCur){clearTimeout(rvT);
      if(now||!rvC.buffer){rvC.buffer=rvIR();rvCur=key()}
      else rvT=setTimeout(function(){if(rvOn&&key()!==rvCur){rvC.buffer=rvIR();rvCur=key()}},140)}
  }
}
var limWk=null;
/* look-ahead brickwall limiter (runs inside AudioWorklet): 5 ms look-ahead, smooth attack, 120 ms release */
function LIMSRC(){
  class Lim extends AudioWorkletProcessor{
    static get parameterDescriptors(){return [{name:'ceil',defaultValue:.96,minValue:.01,maxValue:1,automationRate:'k-rate'}]}
    constructor(){
      super();
      var L=Math.max(32,Math.round(sampleRate*.005)),N=L+1;
      this.L=L;this.N=N;this.w=0;this.t=0;
      this.d=[new Float32Array(N),new Float32Array(N)];
      this.gv=new Float32Array(N).fill(1);
      this.dq=new Float64Array(N+1);this.h=0;this.n=0;
      this.rr=new Float32Array(N).fill(1);this.sum=N;this.r=1;
      this.rc=1-Math.exp(-1/(sampleRate*.12));
    }
    process(ins,outs,pr){
      var I=ins[0],O=outs[0];
      if(!O||!O.length)return true;
      var n=O[0].length,ce=pr.ceil[0],N=this.N,N2=N+1,L=this.L,nc=Math.min(2,O.length),
          w=this.w,t=this.t,gv=this.gv,dq=this.dq,rr=this.rr,d=this.d,i,c,pk,v,g,m,s,o,bi;
      for(i=0;i<n;i++){
        pk=0;
        for(c=0;c<nc;c++){v=(I&&I[c])?I[c][i]:0;d[c][w]=v;v=v<0?-v:v;if(v>pk)pk=v}
        g=pk>ce?ce/pk:1;
        gv[w]=g;
        while(this.n>0){bi=dq[(this.h+this.n-1)%N2];if(gv[bi%N]>=g)this.n--;else break}
        dq[(this.h+this.n)%N2]=t;this.n++;
        while(dq[this.h]<t-L){this.h=(this.h+1)%N2;this.n--}
        m=gv[dq[this.h]%N];
        this.r=m<this.r?m:this.r+(m-this.r)*this.rc;
        this.sum+=this.r-rr[w];rr[w]=this.r;
        s=this.sum/N;
        o=(w+1)%N;
        for(c=0;c<nc;c++)O[c][i]=d[c][o]*s;
        w=(w+1)%N;t++;
        if((t&8191)===0){var q=0,z;for(z=0;z<N;z++)q+=rr[z];this.sum=q}
      }
      this.w=w;this.t=t;
      return true;
    }
  }
  registerProcessor('armamus-lim',Lim);
}
function limLoad(){
  if(limWk||!ac||!ac.audioWorklet||!window.AudioWorkletNode)return;
  var url;
  try{url=URL.createObjectURL(new Blob(['('+LIMSRC.toString()+')()'],{type:'application/javascript'}))}catch(e){return}
  ac.audioWorklet.addModule(url).then(function(){
    var wk=new AudioWorkletNode(ac,'armamus-lim',{numberOfInputs:1,numberOfOutputs:1,outputChannelCount:[2],channelCount:2,channelCountMode:'explicit',channelInterpretation:'speakers'});
    pin.disconnect();pin.connect(wk);wk.connect(ps);limWk=wk;protApply();
  }).catch(function(){}).then(function(){try{URL.revokeObjectURL(url)}catch(e){}});
}
function protBuild(){
  pin=ac.createGain();pc=ac.createDynamicsCompressor();pc.knee.value=0;pc.ratio.value=20;pc.attack.value=.001;pc.release.value=.1;pc.threshold.value=0;
  pgn=ac.createGain();ps=ac.createWaveShaper();ps.oversample='4x';
  /* last-resort soft clip: transparent below .96, rounds off anything above it (no hard edges) */
  var n=2048,cv=new Float32Array(n),i,x,a;
  for(i=0;i<n;i++){x=i*2/(n-1)-1;a=Math.abs(x);cv[i]=a<=.96?x:(x<0?-1:1)*(.96+.04*Math.tanh((a-.96)/.04))}
  ps.curve=cv;pin.connect(pc);pc.connect(pgn);pgn.connect(ps);ps.connect(ac.destination);
  limLoad();
  return pin;
}
function protApply(){
  if(!ac||!pc)return;var t=ac.currentTime,v=prV;
  if(limWk){limWk.parameters.get('ceil').setTargetAtTime(Math.pow(10,-Math.max(.3,v)/20),t,.03);return}
  pc.threshold.setTargetAtTime(v>0?-v:0,t,.03);pc.ratio.setTargetAtTime(v>0?20:1,t,.03);
  pgn.gain.setTargetAtTime(v>0?Math.pow(10,-v*.0285):1,t,.03);
}
/* auto headroom: lower the input by (most of) the biggest EQ boost, so the limiter has little to catch */
function eqHead(){
  if(!eqOn||prV<=0)return 0;
  var mx=0,i,f,d,k;
  for(i=0;i<72;i++){
    f=20*Math.pow(1000,i/71);d=0;
    for(k=0;k<NB;k++)d+=bqdB(k===0?'lowshelf':(k===NB-1?'highshelf':'peaking'),EQF[k],eqg[k],4.3,f);
    if(eqT.lo)d+=bqdB('lowshelf',200,tone*.12,1,f);
    if(eqT.hi)d+=bqdB('highshelf',6000,hi*.12,1,f);
    if(d>mx)mx=d;
  }
  if(eqT.pre)mx+=pre;
  return -Math.min(18,mx)*Math.min(.85,.45+prV*.07);
}
function bqdB(ty,f0,g,Q,f){
  var fs=48000,w0=2*Math.PI*f0/fs,c=Math.cos(w0),sn=Math.sin(w0),A=Math.pow(10,g/40),al,sA,b0,b1,b2,a0,a1,a2;
  if(ty==='peaking'){al=sn/(2*Q);b0=1+al*A;b1=-2*c;b2=1-al*A;a0=1+al/A;a1=-2*c;a2=1-al/A}
  else{al=sn/Math.SQRT2;sA=2*Math.sqrt(A)*al;
    if(ty==='lowshelf'){b0=A*((A+1)-(A-1)*c+sA);b1=2*A*((A-1)-(A+1)*c);b2=A*((A+1)-(A-1)*c-sA);a0=(A+1)+(A-1)*c+sA;a1=-2*((A-1)+(A+1)*c);a2=(A+1)+(A-1)*c-sA}
    else{b0=A*((A+1)+(A-1)*c+sA);b1=-2*A*((A-1)+(A+1)*c);b2=A*((A+1)+(A-1)*c-sA);a0=(A+1)-(A-1)*c+sA;a1=2*((A-1)-(A+1)*c);a2=(A+1)-(A-1)*c-sA}}
  var w=2*Math.PI*f/fs,cw=Math.cos(w),sw=Math.sin(w),c2=Math.cos(2*w),s2=Math.sin(2*w),
  nr=b0+b1*cw+b2*c2,ni=-(b1*sw+b2*s2),dr=a0+a1*cw+a2*c2,di=-(a1*sw+a2*s2);
  return 10*Math.log10((nr*nr+ni*ni)/(dr*dr+di*di));
}
function eqTot(f){
  var d=0;
  EQF.forEach(function(f0,i){d+=bqdB(i===0?'lowshelf':(i===9?'highshelf':'peaking'),f0,eqOn?eqg[i]:0,1.1,f)});
  return d+bqdB('lowshelf',200,eqOn?tone*.12:0,1,f)+bqdB('highshelf',6000,eqOn?hi*.12:0,1,f);
}
function eqCurve(){
  var st=$('#eqstage'),cv=$('#eqcv');if(!st)return;
  var w=st.clientWidth,h=st.clientHeight;if(!w||!h)return;
  var dpr=window.devicePixelRatio||1;cv.width=Math.round(w*dpr);cv.height=Math.round(h*dpr);
  var c=cv.getContext('2d');c.scale(dpr,dpr);
  var sr=st.getBoundingClientRect(),pts=[],bt=h;
  Array.prototype.forEach.call($('#eqb').children,function(d){
    var k=d.querySelector('.kn').getBoundingClientRect(),r=d.querySelector('.rl').getBoundingClientRect();
    pts.push([k.left+k.width/2-sr.left,k.top+k.height/2-sr.top]);bt=r.bottom-sr.top});
  if(pts.length<2)return;
  var on=eqOn,ac1=on?'rgba(190,165,255,.95)':'rgba(255,255,255,.3)',n=pts.length,i;
  function path(){c.beginPath();c.moveTo(pts[0][0],pts[0][1]);
    for(i=0;i<n-1;i++){var p0=pts[Math.max(0,i-1)],p1=pts[i],p2=pts[i+1],p3=pts[Math.min(n-1,i+2)];
      c.bezierCurveTo(p1[0]+(p2[0]-p0[0])/6,p1[1]+(p2[1]-p0[1])/6,p2[0]-(p3[0]-p1[0])/6,p2[1]-(p3[1]-p1[1])/6,p2[0],p2[1])}}
  path();c.lineTo(pts[n-1][0],bt);c.lineTo(pts[0][0],bt);c.closePath();
  var g=c.createLinearGradient(0,0,0,bt);g.addColorStop(0,on?'rgba(150,115,255,.42)':'rgba(255,255,255,.14)');g.addColorStop(1,'rgba(150,115,255,0)');
  c.fillStyle=g;c.fill();
  c.strokeStyle=ac1;c.lineWidth=2;c.lineCap='round';
  pts.forEach(function(q){c.beginPath();c.moveTo(q[0],q[1]);c.lineTo(q[0],bt);c.stroke()});
  path();c.lineJoin='round';c.stroke();
}
window.addEventListener('resize',function(){if($('#eq').classList.contains('on'))eqCurve()});
